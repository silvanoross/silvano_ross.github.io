Schematic

  1. Process and purify soluble HLA class I using immunoaffinity purification
  2. Fractionate using HP/LC
  3. Run on TOF Mass Spec
  4. Use mass spec analysis software to obtain peptide sequences, source proteins, and genes
  5. Perform analysis

# select for top genes based on sample area
top_area_genes <- pep_clean |> 
  select(Gene,accession_clean,Area.Sample.1) |>
  arrange(desc(Area.Sample.1)) |>
  head(11)

genes <- top_area_genes$Gene

# get pathway from API for Gene column using gprofiler
gostres <- gost(query = genes,
                organism = "hsapiens",
                sources = c("GO:BP", "GO:MF", "GO:CC", "KEGG", "REAC"))
# add pathways to df
